async function importar(dir: string, archivo: string, e: EntradaLibro, modo: ModoAplicacion) {
  const hojas = await leerLibroCompleto(readFileSync(join(OUT, dir, archivo)));
  const libro = leerLibro(hojas);
  const control = validarControl(libro.control, { ciclo: e.ciclo.id, instancia: e.instancia, plantilla: e.plantilla, huella: huellaDe(e.plantilla, e.secciones) });
  const propuestas = clasificar(libro.bloques, destinosDe(e.secciones), libro.control);
  const r = aplicar({ bloques: libro.bloques, propuestas, contexto: ctx(e), modo, estado: "draft", rol: "jefe", aporta: libro.control?.valores.equipo });
  return { hojas, libro, control, propuestas, r };
}

describe("verificar los libros generados", () => {
  it("todos los .xlsx se abren con exceljs y traen hojas ocultas de control", async () => {
    const { default: ExcelJS } = await import("exceljs");
    const todos = Object.values(D).flatMap((d) => listar(d).map((f) => join(OUT, d, f)));
    expect(todos.length).toBeGreaterThanOrEqual(14);
    for (const ruta of todos) {
      const wb = new ExcelJS.Workbook();
      await wb.xlsx.readFile(ruta);
      expect(wb.worksheets.length).toBeGreaterThan(0);
      if (/04-|01-|02-|03-|05-/.test(ruta)) {
        const ocultas = wb.worksheets.filter((w) => w.state !== "visible").map((w) => w.name);
        expect(ocultas).toEqual(["Listas", "_Control"]);
      }
    }
  });

  it("el libro vacío, el diligenciado y los de equipo entran sin errores", async () => {
    const vacio = base();
    const lleno = base({ contenidos: CONT_LLENO, anexos: ANEXOS });
    const rv = await importar(D.vacio, nombreArchivoLibro(vacio), vacio, "agregar");
    expect(rv.control.error).toBeUndefined();
    expect(rv.control.avisos).toEqual([]);
    expect(rv.propuestas.every((p) => p.origen === "control" && p.nivel === "reconocida")).toBe(true);
    // Sin diligenciar solo avisa de lo que falta o de las filas de ejemplo; nunca de errores de formato.
    expect(rv.r.avisos.every((a) => /no hay texto para importar|de ejemplo se omiti|falta «/.test(a))).toBe(true);
    bitacora.vacio = { avisos: rv.r.avisos, totales: rv.r.totales, resumen: rv.r.resumen, hojas: rv.libro.bloques.length, secciones: Object.keys(rv.r.contenidos), contenidos: rv.r.contenidos };

    for (const modo of ["agregar", "vacios", "reemplazar"] as ModoAplicacion[]) {
      const rl = await importar(D.lleno, nombreArchivoLibro(lleno), base({ contenidos: {}, anexos: ANEXOS }), modo);
      expect(rl.control.error).toBeUndefined();
      expect(rl.r.avisos.filter((a) => !/^Las evidencias no se reemplazan/.test(a))).toEqual([]);
      expect(rl.r.totales.conAviso).toBe(0);
      expect(Object.keys(rl.r.contenidos).length).toBeGreaterThan(4);
      bitacora[`lleno_${modo}`] = { resumen: rl.r.resumen, totales: rl.r.totales, secciones: rl.r.secciones.map((s) => ({ t: s.titulo, ag: s.agregadas, comp: s.completadas, om: s.omitidas, li: s.listas })) };
    }
    // Mismo libro sobre un informe que ya trae lo mismo: no agrega nada nuevo (omitidas).
    const ya = await importar(D.lleno, nombreArchivoLibro(lleno), lleno, "agregar");
    bitacora.lleno_sobre_lleno = { resumen: ya.r.resumen, secciones: ya.r.secciones.map((s) => ({ t: s.titulo, ag: s.agregadas, om: s.omitidas })) };

    for (const f of listar(D.equipos)) {
      const re = await importar(D.equipos, f, base({ anexos: ANEXOS }), "agregar");
      expect(re.control.error).toBeUndefined();
      expect(re.r.avisos.every((a) => /no hay texto para importar|de ejemplo se omiti|falta «/.test(a))).toBe(true);
      expect(re.propuestas.every((p) => p.origen === "control")).toBe(true);
      bitacora[`equipo_${f}`] = { hojas: re.libro.bloques.map((b) => b.hoja), equipo: re.libro.control?.valores.equipo, avisos: re.r.avisos };
    }
  });

  it("el libro con errores muestra avisos y el de otro informe se rechaza", async () => {
    const e = base({ contenidos: CONT_LLENO, anexos: ANEXOS });
    const rr = await importar(D.errores, nombreArchivoLibro(e), base({ anexos: ANEXOS }), "agregar");
    expect(rr.control.error).toBeUndefined();
    expect(rr.r.avisos.length).toBeGreaterThan(10);
    expect(rr.r.totales.conAviso + rr.r.totales.sinImportar).toBeGreaterThan(5);
    const soportes = rr.propuestas[rr.libro.bloques.findIndex((b) => b.hoja === "Mis soportes")];
    bitacora.errores = { avisos: rr.r.avisos, totales: rr.r.totales, resumen: rr.r.resumen, soportes: { origen: soportes.origen, nivel: soportes.nivel, puntaje: soportes.puntaje }, propuestas: rr.propuestas.map((p, i) => ({ hoja: rr.libro.bloques[i].hoja, destino: p.destino, origen: p.origen, nivel: p.nivel })) };

    const o = base({ instancia: "inst-otra-unidad-2026", unidad: "Facultad de Salud" });
    const ro = await importar(D.otro, nombreArchivoLibro(o), base(), "agregar");
    expect(ro.control.error).toMatch(/no corresponde a este informe/);
    bitacora.otro = { error: ro.control.error };
    const cambiada = validarControl(ro.libro.control, { ciclo: "ciclo-2026", instancia: "inst-otra-unidad-2026", plantilla: "A", huella: "00000000" });
    bitacora.huella = cambiada.avisos;
    bitacora.noEsLibro = validarControl({ valores: { formato: "otro" }, hojas: {} }, { ciclo: "x", instancia: "x", plantilla: "A", huella: "x" });
    bitacora.otroCiclo = validarControl({ valores: { formato: "libro-informe", ciclo: "c-2025", anio: "2025", instancia: "inst-ing-2026" }, hojas: {} }, { ciclo: "ciclo-2026", instancia: "inst-ing-2026", plantilla: "A", huella: "x" });
    bitacora.otraPlantilla = validarControl({ valores: { formato: "libro-informe", plantilla: "B", instancia: "inst-ing-2026" }, hojas: {} }, { ciclo: "ciclo-2026", instancia: "inst-ing-2026", plantilla: "A", huella: "x" });
    // Informe enviado: no se escribe.
    const hojas = await leerLibroCompleto(readFileSync(join(OUT, D.lleno, nombreArchivoLibro(e))));
    const libro = leerLibro(hojas);
    const env = aplicar({ bloques: libro.bloques, propuestas: clasificar(libro.bloques, destinosDe(A), libro.control), contexto: ctx(base()), modo: "agregar", estado: "sent", rol: "jefe" });
    bitacora.enviado = env.bloqueado;
  });

  it("las plantillas de filas, resultados y datos requeridos se leen con su lector", async () => {
    const { default: ExcelJS } = await import("exceljs");
    const leerHojaPlana = async (ruta: string, nombre: string) => {
      const wb = new ExcelJS.Workbook();
      await wb.xlsx.readFile(ruta);
      const ws = wb.getWorksheet(nombre)!;
      const filas: string[][] = [];
      ws.eachRow({ includeEmpty: false }, (row) => { const c: string[] = []; row.eachCell({ includeEmpty: true }, (cel, n) => { c[n - 1] = String(cel.value ?? ""); }); filas.push(Array.from(c, (x) => x ?? "")); });
      return filas;
    };
    const f1 = await leerHojaPlana(join(OUT, D.filas, "plantilla-puntos-criticos.xlsx"), "Filas");
    const r1 = leerFilasSeccion(A.find((s) => s.tipo === "filas")!.columnas!, f1, CRITERIOS_BASE.map((d) => d.nombre));
    bitacora.filas_puntos = r1;
    expect(r1.filas.length).toBe(2);
    const f2 = await leerHojaPlana(join(OUT, D.filas, "plantilla-logros.xlsx"), "Filas");
    bitacora.filas_logros = leerFilasSeccion([...COLUMNAS_LOGROS], f2);
    const f3 = await leerHojaPlana(join(OUT, D.admin, "datos-requeridos.xlsx"), "Datos requeridos");
    const r3 = leerFilasRequeridos(f3, "ciclo-2026", [...CRITERIOS_BASE, ...APROBADOS], { A: 0, B: 0, C: 0, D: 0 });
    expect(r3.datos.length).toBe(5);
    expect(r3.avisos).toEqual([]);
    writeFileSync(join(SCRATCH, "bitacora-excel.json"), JSON.stringify(bitacora, null, 1));
  });
});
